# --- Provider API keys: sourced by a shell, NOT EnvironmentFile --------------
# HelixAgent scans ~137 candidate provider env-var names at startup; with none
# present it discovers 0 providers and every completion returns
# 503 no_provider_available. The operator's keys live in $HOME/api_keys.sh
# (override with CMA_KEYS_FILE).
#
# That file CANNOT be used as an EnvironmentFile=. Measured 2026-09-03:
#   * 88 of its lines are `export KEY=value`, 0 are bare `KEY=value`. systemd's
#     EnvironmentFile parser does not execute shell, so it reads the name as
#     "export KEY" — not a valid variable name — and DROPS the line.
#   * 41 lines interpolate another variable (`KEY=$ApiKey_Foo`). systemd does no
#     shell expansion in EnvironmentFile, so those would arrive as literal
#     "$ApiKey_Foo" strings even if the names parsed.
# Pointing EnvironmentFile= at it would therefore fail SILENTLY — the unit would
# start clean and still discover 0 providers.
#
# So the file is SOURCED by a shell that then execs the binary. `set -a` exports
# every assignment; `exec` replaces the shell so systemd still supervises the
# real process (no stray wrapper PID, MAINPID stays correct). `bash -c` (not
# `-lc`) avoids pulling in interactive profile side effects. The `[ -f ]` guard
# keeps a missing keys file from failing the unit — it degrades to zero
# providers, which the log states plainly.
#
# The keys file is NOT modified, NOT copied, and no value is ever logged.
# exec-with-api-keys.sh sources the keys file, then execs the binary in place
# (so systemd supervises helixagent directly, MAINPID intact). A tracked
# script rather than an inline `bash -c` because the inline form needs
# systemd's `$$` escaping for every literal dollar, which `systemctl show`
# renders identically whether the escape is right or wrong, and which cannot
# be tested outside systemd. This script is testable by hand.
ExecStart=/home/milosvasic/Projects/helix_code/scripts/systemd/exec-with-api-keys.sh /home/milosvasic/Projects/helix_code/submodules/helix_agent/bin/helixagent

[Service]
Type=exec
# /home/milosvasic/opt/llamacpp_gpu/current/llama-server / 99 are resolved by
# scripts/install_systemd_units.sh, which PROBES each candidate binary with
# `llama-server --list-devices` and picks the first that actually enumerates a
# GPU (§11.4.201: assert the real condition, not the proxy "a path exists").
#
# WHY A NON-DISTRO BINARY IS PREFERRED HERE, measured 2026-09-03:
# /usr/bin/llama-server (8681-Debian) is a CPU-ONLY build. It ships only
# libggml-cpu-*.so under /usr/lib/x86_64-linux-gnu/ggml/backends0 — there is no
# CUDA or Vulkan backend in the archive at all (`apt-cache search ggml` offers
# blas/hip/vulkan but NO cuda backend package) — so `--list-devices` prints an
# empty device list and the host's RTX 3060 is unreachable. Upstream ggml-org
# publishes NO Linux CUDA prebuilt either (CUDA assets are win-* only), so the
# GPU-capable binary installed at ~/opt/llamacpp_gpu/current is the upstream
# ubuntu-vulkan-x64 build, which reaches the 3060 through the NVIDIA Vulkan ICD
# and its NV_coopmat2 cooperative-matrix (tensor-core) path.
#
# The Debian package is deliberately left INSTALLED and UNTOUCHED as the
# fallback candidate (§11.4.122: no capability removed) — if the GPU build ever
# goes missing the installer resolves back to /usr/bin/llama-server with
# 99=0 rather than installing a unit that claims an offload it cannot
# perform.
#
# The upstream tarball is self-contained: its libllama/libggml* resolve via
# rpath from the binary's own directory, verified from an unrelated cwd with no
# LD_LIBRARY_PATH set. No library env var is therefore needed here, and NOT
# setting one is what keeps the two installs from cross-loading each other's
# ABI-incompatible backends.
#
# Measured on this host, same model, same binary, only -ngl varying:
#   -ngl 0  (CPU) : pp128  200.73 t/s | tg32   0.74 t/s   (at load average 83)
#   -ngl 99 (GPU) : pp512 3728.42 t/s | tg64 108.75 t/s
# Generation is ~147x faster than the contended CPU path, and ~11-54x faster
# than the 2-10 t/s the CPU path delivered at lower load. The GPU path is also
# largely immune to the CPU contention that makes the CPU figure so variable.
#
# Threads stay at 8 of 16 deliberately: this is a shared host. With all layers
# offloaded the CPU does little more than sampling and detokenisation, so 8 is
# generous headroom rather than a throughput limit.

=== llmsverifier.service unit file ===
# /home/milosvasic/.config/systemd/user/llmsverifier.service
[Unit]
Description=LLMsVerifier (model/provider verification + scoring API) on :8100
Documentation=file:///home/milosvasic/Projects/helix_code/docs/SYSTEMD.md
# Ordered before helixllm-gateway: the gateway polls this service's
# /api/scores on an interval and falls back to a static score table whenever it
# is unreachable (CONST-036 makes this service the single source of truth for
# model/provider metadata and scoring, so the static table is a degraded mode).
After=network-online.target
Wants=network-online.target
PartOf=helix.target

[Service]
Type=simple
Environment=PATH=/usr/local/bin:/usr/bin:/bin

# Port 8100 is the address HelixLLM's gateway already looks for
# (HELIX_LLM_VERIFIER_URL, default http://localhost:8100). The binary's own
# default is 8080, which collides with nothing here but is not where any
# consumer looks -- so the port is passed explicitly rather than left to differ.
#
# NOTE: 8100 appears in this submodule's own sources as "HelixAgentPort" (see
# pkg/cliagents/generator.go). That is a separate, unrelated use of the same
# number for a different service; HelixAgent actually runs on 7061 here. The
# binding below is the consumer-facing contract, not that constant.

# The API JWT secret lives in .env (mode 0600, gitignored) -- never inlined into
# this unit, which is world-readable and version-controlled (CONST-042 / §12.1).
# Without it the binary logs "Using default JWT secret" and runs with the
# placeholder committed in config.yaml.
EnvironmentFile=-/home/milosvasic/Projects/helix_code/.env

# WorkingDirectory is load-bearing: the binary resolves its default `-c
# config.yaml` and its SQLite path (./llm-verifier.db, per that config)
# relative to the working directory. Starting it elsewhere silently produces a
# second, empty database in whatever directory systemd happened to use.
WorkingDirectory=/home/milosvasic/Projects/helix_code/submodules/llms_verifier/llm-verifier
#
# -c points at OUR config, not the submodule's.
#
# The submodule ships a tracked config.yaml with `llms: []`. Adding this
# platform's provider endpoints to that file would inject consuming-project
# context into a reusable submodule (§11.4.28) AND be overwritten on the next
# submodule update. config/llmsverifier/config.yaml is owned by this repo; the
# server seeds its provider/model rows from that file's `llms:` at boot
# (idempotent), so the submodule stays project-agnostic.
ExecStart=/home/milosvasic/Projects/helix_code/submodules/llms_verifier/llm-verifier/bin/llm-verifier \
    -c /home/milosvasic/Projects/helix_code/config/llmsverifier/config.yaml \
    server --port 8100

Restart=on-failure
RestartSec=10
LimitNOFILE=65536
TimeoutStartSec=120

[Install]
WantedBy=helix.target
